1//! vb6-run: Standalone Visual Basic 6.0 PE Runner and Inspector for Linux / macOS.
2//!
3//! Loads legacy Win32 VB6 executables into executable memory, resolves imports
4//! against msvbvm60 and Win32 shims, and dispatches execution directly.
5
6use std::env;
7use std::fs;
8use std::path::Path;
9use std::process;
10
11use msvbvm60::pe::loader::LoadedPe;
12use msvbvm60::types::ProjectInfo;
13
14/// Prints the application banner to standard output.
15fn print_banner() {
16    println!("╔═══════════════════════════════════════════════════════════════╗");
17    println!("║       VB6-RUN: Visual Basic 6.0 Linux / Native Runner        ║");
18    println!("║             Powered by Native Rust MSVBVM60 Runtime           ║");
19    println!("╚═══════════════════════════════════════════════════════════════╝");
20}
21
22/// Prints command-line syntax and usage options to standard output.
23///
24/// # Arguments
25/// * `prog_name` - Program invocation name (typically `argv[0]`).
26fn print_usage(prog_name: &str) {
27    println!("\nUsage: {} <path-to-vb6-exe> [options]", prog_name);
28    println!("\nOptions:");
29    println!("  --inspect     Inspect PE structure, sections, and metadata without running");
30    println!("  --dump-info   Dump full VBHeader, ProjectInfo, and object tables");
31    println!("  --run         Execute the binary (default behavior)");
32    println!("  --help, -h    Display this help message\n");
33}
34
35/// CLI entry point for loading and running Visual Basic 6 executables on Linux/macOS.
36fn main() {
37    let args: Vec<String> = env::args().collect();
38    if args.len() < 2 {
39        print_banner();
40        print_usage(&args[0]);
41        process::exit(1);
42    }
43
44    let mut exe_path: Option<&str> = None;
45    let mut inspect_only = false;
46    let mut dump_info = false;
47
48    for arg in &args[1..] {
49        match arg.as_str() {
50            "--help" | "-h" => {
51                print_banner();
52                print_usage(&args[0]);
53                process::exit(0);
54            }
55            "--inspect" => inspect_only = true,
56            "--dump-info" => dump_info = true,
57            "--run" => {}
58            other if !other.starts_with("--") => exe_path = Some(other),
59            unknown => {
60                eprintln!("Unknown option: {}", unknown);
61                print_usage(&args[0]);
62                process::exit(1);
63            }
64        }
65    }
66
67    let file_path = match exe_path {
68        Some(p) => p,
69        None => {
70            eprintln!("Error: No executable file path specified.");
71            print_usage(&args[0]);
72            process::exit(1);
73        }
74    };
75
76    if !Path::new(file_path).exists() {
77        eprintln!("Error: File not found: '{}'", file_path);
78        process::exit(1);
79    }
80
81    print_banner();
82    println!("[*] Reading binary: {}", file_path);
83
84    let data = match fs::read(file_path) {
85        Ok(bytes) => bytes,
86        Err(e) => {
87            eprintln!("Error reading file: {}", e);
88            process::exit(1);
89        }
90    };
91
92    println!("[*] Parsing PE32 and mapping sections ({} bytes)...", data.len());
93    let loaded_pe = match LoadedPe::load(&data) {
94        Ok(pe) => pe,
95        Err(e) => {
96            eprintln!("PE Loader Error: {}", e);
97            process::exit(1);
98        }
99    };
100
101    println!("[+] PE32 successfully mapped into virtual memory:");
102    println!("    - Preferred Base: 0x{:08X}", loaded_pe.preferred_base);
103    println!("    - Allocated Base: {:p}", loaded_pe.mapped_base);
104    println!("    - Relocation Delta: 0x{:X}", loaded_pe.delta);
105    println!("    - Entry Point RVA: 0x{:08X}", loaded_pe.entry_point_rva);
106    println!("    - Sections Count: {}", loaded_pe.sections.len());
107
108    for (i, sec) in loaded_pe.sections.iter().enumerate() {
109        println!(
110            "      [{}] {:<8} VA: 0x{:08X} (VSize: 0x{:06X}, RawSize: 0x{:06X})",
111            i, sec.name, sec.virtual_address, sec.virtual_size, sec.size_of_raw_data
112        );
113    }
114
115    println!("\n[*] Scanning for Visual Basic 6 metadata ('VB5!' signature)...");
116    let vb_hdr_ptr = loaded_pe.find_vb_header();
117
118    match vb_hdr_ptr {
119        Some(ptr) => {
120            let vb_hdr = unsafe { *ptr };
121            println!("[+] Visual Basic 6 Executable Verified!");
122            let build = vb_hdr.runtime_build_version;
123            let rev = vb_hdr.runtime_revision;
124            let forms = vb_hdr.w_forms_count;
125            let threads = vb_hdr.dw_thread_count;
126            let sub_main = vb_hdr.lp_sub_main;
127            println!("    - Build Version: 0x{:04X}, Revision: 0x{:04X}", build, rev);
128            println!("    - Forms Count: {}", forms);
129            println!("    - Thread Count: {}", threads);
130            println!("    - Sub Main Address: 0x{:08X}", sub_main);
131
132            let base = loaded_pe.mapped_base as usize;
133            let size = loaded_pe.mapped_size;
134            let is_valid_ptr = |ptr: u32| -> bool {
135                let p = ptr as usize;
136                p >= base && p < base + size
137            };
138
139            if is_valid_ptr(vb_hdr.lp_project_data) {
140                let proj_info = unsafe { &*(vb_hdr.lp_project_data as *const ProjectInfo) };
141                if is_valid_ptr(proj_info.lp_project_name) {
142                    let name_cstr = unsafe { std::ffi::CStr::from_ptr(proj_info.lp_project_name as *const i8) };
143                    println!("    - Project Name: {}", name_cstr.to_string_lossy());
144                }
145                if is_valid_ptr(proj_info.lp_project_title) {
146                    let title_cstr = unsafe { std::ffi::CStr::from_ptr(proj_info.lp_project_title as *const i8) };
147                    println!("    - Project Title: {}", title_cstr.to_string_lossy());
148                }
149                if is_valid_ptr(proj_info.lp_project_description) {
150                    let desc_cstr = unsafe { std::ffi::CStr::from_ptr(proj_info.lp_project_description as *const i8) };
151                    println!("    - Description: {}", desc_cstr.to_string_lossy());
152                }
153            }
154
155            if dump_info {
156                println!("\n--- [VBHeader Detailed Dump] ---");
157                println!("  Magic: {:?}", std::str::from_utf8(&vb_hdr.sz_magic).unwrap_or("????"));
158                println!("  Language DLL: {:?}", std::str::from_utf8(&vb_hdr.sz_language_dll).unwrap_or(""));
159                println!("  LCID: 0x{:08X}", { vb_hdr.lcid });
160                println!("  Thread Flags: 0x{:08X}", { vb_hdr.dw_thread_flags });
161                println!("  GUI Table: 0x{:08X}", { vb_hdr.lp_gui_table });
162                println!("  Import Table: 0x{:08X}", { vb_hdr.lp_import_table });
163                println!("  Export Table: 0x{:08X}", { vb_hdr.lp_export_table });
164
165                if vb_hdr.lp_project_data != 0 {
166                    let proj_info = unsafe { &*(vb_hdr.lp_project_data as *const ProjectInfo) };
167                    println!("\n--- [ProjectInfo Dump] ---");
168                    println!("  Version: 0x{:08X}", { proj_info.dw_version });
169                    println!("  ObjectTable Pointer: 0x{:08X}", { proj_info.lp_object_table });
170                    println!("  Code Depot: 0x{:08X}", { proj_info.lp_code_depot });
171
172                    if proj_info.lp_object_table != 0 {
173                        let obj_tbl = unsafe { &*(proj_info.lp_object_table as *const msvbvm60::types::ObjectTable) };
174                        println!("\n--- [ObjectTable Dump] ---");
175                        println!("  Total Objects: {}", { obj_tbl.dw_total_objects_count });
176                        println!("  Compiled Objects: {}", { obj_tbl.dw_compiled_objects_count });
177                        println!("  Objects in Use: {}", { obj_tbl.dw_objects_in_use_count });
178                    }
179                }
180            }
181
182            if inspect_only || dump_info {
183                println!("\n[+] Inspection finished successfully.");
184                return;
185            }
186
187            println!("\n[*] Starting execution via ThunRTMain...");
188            let exit_code = match loaded_pe.execute() {
189                Ok(code) => code,
190                Err(e) => {
191                    eprintln!("Execution error: {}", e);
192                    process::exit(1);
193                }
194            };
195
196            println!("\n[+] Execution completed with return code: {}", exit_code);
197            process::exit(exit_code);
198        }
199        None => {
200            println!("[-] No 'VB5!' header signature found. (Is this a standard C/C++ PE?)");
201            if inspect_only {
202                println!("[+] PE inspection completed.");
203                return;
204            }
205            eprintln!("Cannot execute: binary is not a recognized Visual Basic 6 executable.");
206            process::exit(1);
207        }
208    }
209}
