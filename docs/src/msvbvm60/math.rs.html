39// ----------------------------------------------------------------------------
40// 64-bit Integer Multiplication Runtime Helper
41// ----------------------------------------------------------------------------
42
43/// 64-bit integer multiplication runtime helper emitted by MSVC backends for 32-bit x86 targets.
44///
45/// Multiplies two 64-bit integers with wrapping semantics.
46///
47/// # Arguments
48/// * `a` - First 64-bit operand.
49/// * `b` - Second 64-bit operand.
50///
51/// # Returns
52/// The 64-bit result of `a * b` with wrapping on overflow.
53#[no_mangle]
54pub unsafe extern "C" fn _allmul(a: i64, b: i64) -> i64 {
55    a.wrapping_mul(b)
56}
