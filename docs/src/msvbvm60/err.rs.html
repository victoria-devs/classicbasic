1//! Structured Error Handling and `ErrObject` implementation.
2//!
3//! Provides the runtime state machine for Visual Basic 6 structured error handling
4//! (`On Error Resume Next`, `On Error GoTo <Label>`, `On Error GoTo 0`), runtime
5//! error code lookup, and the intrinsic COM `ErrObject` (`Err.Number`, `Err.Description`,
6//! `Err.Source`, `Err.Clear`, `Err.Raise`, `Err.LastDllError`).

13/// Internal thread/global state for the Visual Basic 6 error handling engine.
14#[derive(Debug, Clone)]
15pub struct VbErrorState {
16    /// Visual Basic runtime error number (e.g., 11 for Division by zero, 0 for no error).
17    pub number: i32,
18    /// Source of the error (e.g., project or class name).
19    pub source: String,
20    /// Human-readable description of the error.
21    pub description: String,
22    /// Path to the Help file associated with this error.
23    pub help_file: String,
24    /// Context ID in the Help file.
25    pub help_context: i32,
26    /// Error code returned by the last Win32 API call (`GetLastError`).
27    pub last_dll_error: i32,
28    /// Current error handler mode:
29    /// - `0`: No active handler / `On Error GoTo 0` (unhandled exceptions abort).
30    /// - `-1`: `On Error Resume Next` (ignore error and continue next instruction).
31    /// - `> 0`: `On Error GoTo <Label>` (jump to specified label/block).
32    pub handler_mode: i32,
33}
34
35impl Default for VbErrorState {
36    fn default() -> Self {
37        Self {
38            number: 0,
39            source: String::new(),
40            description: String::new(),
41            help_file: String::new(),
42            help_context: 0,
43            last_dll_error: 0,
44            handler_mode: 0,
45        }
46    }
47}
48
49/// Global error state tracked across invocations.
50static GLOBAL_ERROR_STATE: Mutex<Option<VbErrorState>> = Mutex::new(None);
51
52/// Ensures error state is initialized and returns a locked guard.
53fn with_error_state<F, R>(f: F) -> R
54where
55    F: FnOnce(&mut VbErrorState) -> R,
56{
57    let mut guard = GLOBAL_ERROR_STATE.lock().unwrap();
58    if guard.is_none() {
59        *guard = Some(VbErrorState::default());
60    }
61    f(guard.as_mut().unwrap())
62}

64/// Returns the standard Visual Basic 6 error description for a given error code.
65pub fn get_standard_error_description(code: i32) -> &'static str {
66    match code {
67        3 => "Return without GoSub",
68        5 => "Invalid procedure call or argument",
69        6 => "Overflow",
70        7 => "Out of memory",
71        9 => "Subscript out of range",
72        10 => "This array is fixed or temporarily locked",
73        11 => "Division by zero",
74        13 => "Type mismatch",
75        14 => "Out of string space",
76        20 => "Resume without error",
77        28 => "Out of stack space",
78        52 => "Bad file name or number",
79        53 => "File not found",
80        54 => "Bad file mode",
81        55 => "File already open",
82        57 => "Device I/O error",
83        58 => "File already exists",
84        61 => "Disk full",
85        62 => "Input past end of file",
86        67 => "Too many files",
87        68 => "Device unavailable",
88        70 => "Permission denied",
89        75 => "Path/File access error",
90        76 => "Path not found",
91        91 => "Object variable or With block variable not set",

95        429 => "ActiveX component can't create object",
96        438 => "Object doesn't support this property or method",
97        440 => "Automation error",
98        1001 => "Custom Raised Business Error",

103/// Clears the current runtime error state (`Err.Clear`).
104pub fn clear_vb_error() {
105    with_error_state(|state| {
106        state.number = 0;
107        state.source.clear();
108        state.description.clear();
109        state.help_file.clear();
110        state.help_context = 0;
111    });
112}
113
114/// Sets the current runtime error state.
115pub fn set_vb_error(number: i32, description: Option<&str>, source: Option<&str>) {
116    with_error_state(|state| {
117        state.number = number;
118        state.description = match description {
119            Some(d) => d.to_string(),
120            None => get_standard_error_description(number).to_string(),
121        };
122        state.source = match source {
123            Some(s) => s.to_string(),
124            None => String::new(),
125        };
126    });
127}

160/// COM VTable for Visual Basic 6 `_ErrObject` interface.
161///
162/// Matches the exact dual-interface VTable layout expected by MSVBVM60-compiled binaries:
163/// - Slots 0..2: `IUnknown` (`QueryInterface`, `AddRef`, `Release`)
164/// - Slots 3..6: `IDispatch` (`GetTypeInfoCount`, `GetTypeInfo`, `GetIDsOfNames`, `Invoke`)
165/// - Slot 7  (0x1C): `get_Number`
166/// - Slot 8  (0x20): `put_Number`
167/// - Slot 9  (0x24): `get_Source`
168/// - Slot 10 (0x28): `put_Source`
169/// - Slot 11 (0x2C): `get_Description`
170/// - Slot 12 (0x30): `put_Description`
171/// - Slot 13 (0x34): `get_HelpFile`
172/// - Slot 14 (0x38): `put_HelpFile`
173/// - Slot 15 (0x3C): `get_HelpContext`
174/// - Slot 16 (0x40): `put_HelpContext`
175/// - Slot 17 (0x44): `Raise`
176/// - Slot 18 (0x48): `Clear`
177/// - Slot 19 (0x4C): `get_LastDllError`
178#[repr(C)]
179pub struct ErrObjectVTable {
180    // IUnknown
181    pub QueryInterface: unsafe extern "system" fn(this: *mut c_void, riid: *const c_void, ppv: *mut *mut c_void) -> i32,
182    pub AddRef: unsafe extern "system" fn(this: *mut c_void) -> u32,
183    pub Release: unsafe extern "system" fn(this: *mut c_void) -> u32,
184
185    // IDispatch
186    pub GetTypeInfoCount: unsafe extern "system" fn(this: *mut c_void, pctinfo: *mut u32) -> i32,
187    pub GetTypeInfo: unsafe extern "system" fn(this: *mut c_void, iTInfo: u32, lcid: u32, ppTInfo: *mut *mut c_void) -> i32,
188    pub GetIDsOfNames: unsafe extern "system" fn(this: *mut c_void, riid: *const c_void, rgszNames: *const *const u16, cNames: u32, lcid: u32, rgDispId: *mut i32) -> i32,
189    pub Invoke: unsafe extern "system" fn(this: *mut c_void, dispIdMember: i32, riid: *const c_void, lcid: u32, wFlags: u16, pDispParams: *mut c_void, pVarResult: *mut c_void, pExcepInfo: *mut c_void, puArgErr: *mut u32) -> i32,
190
191    // _ErrObject methods
192    pub get_Number: unsafe extern "system" fn(this: *mut c_void, p_val: *mut i32) -> i32,
193    pub put_Number: unsafe extern "system" fn(this: *mut c_void, val: i32) -> i32,
194    pub get_Source: unsafe extern "system" fn(this: *mut c_void, pbstr: *mut Bstr) -> i32,
195    pub put_Source: unsafe extern "system" fn(this: *mut c_void, bstr: Bstr) -> i32,
196    pub get_Description: unsafe extern "system" fn(this: *mut c_void, pbstr: *mut Bstr) -> i32,
197    pub put_Description: unsafe extern "system" fn(this: *mut c_void, bstr: Bstr) -> i32,
198    pub get_HelpFile: unsafe extern "system" fn(this: *mut c_void, pbstr: *mut Bstr) -> i32,
199    pub put_HelpFile: unsafe extern "system" fn(this: *mut c_void, bstr: Bstr) -> i32,
200    pub get_HelpContext: unsafe extern "system" fn(this: *mut c_void, p_val: *mut i32) -> i32,
201    pub put_HelpContext: unsafe extern "system" fn(this: *mut c_void, val: i32) -> i32,
202    pub Raise: unsafe extern "system" fn(
203        this: *mut c_void,
204        number: i32,
205        source: *const Variant,
206        desc: *const Variant,
207        helpfile: *const Variant,
208        helpcontext: *const Variant,
209    ) -> i32,
210    pub Clear: unsafe extern "system" fn(this: *mut c_void) -> i32,
211    pub get_LastDllError: unsafe extern "system" fn(this: *mut c_void, p_val: *mut i32) -> i32,
212}

335unsafe extern "system" fn err_Raise(
336    _this: *mut c_void,
337    number: i32,
338    source: *const Variant,
339    desc: *const Variant,
340    _helpfile: *const Variant,
341    _helpcontext: *const Variant,
342) -> i32 {
343    let mut custom_desc: Option<String> = None;
344    let mut custom_source: Option<String> = None;
345
346    if !desc.is_null() {
347        if let Some(b) = (*desc).as_bstr() {
348            if !b.is_null() {
349                custom_desc = Some(bstr_to_string(b));
350            }
351        }
352    }
353
354    if !source.is_null() {
355        if let Some(b) = (*source).as_bstr() {
356            if !b.is_null() {
357                custom_source = Some(bstr_to_string(b));
358            }
359        }
360    }
361
362    set_vb_error(
363        number,
364        custom_desc.as_deref(),
365        custom_source.as_deref(),
366    );
367    S_OK
368}

421/// Sets or updates the active exception handling mode (`__vbaOnError`, Ordinal 352).
422///
423/// # Arguments
424/// * `handler` - Handler target address or control flag:
425///   - `-1` (`0xFFFFFFFF`): `On Error Resume Next`
426///   - `0`: `On Error GoTo 0` (reset error handler)
427///   - `> 0`: `On Error GoTo <Label>`
428#[no_mangle]
429pub unsafe extern "system" fn __vbaOnError(handler: *mut c_void) {
430    let val = handler as usize as isize;
431    if val == -1 {
432        set_vb_error_handler(-1);
433    } else if val == 0 {
434        set_vb_error_handler(0);
435    } else {
436        set_vb_error_handler(1);
437    }
438}
