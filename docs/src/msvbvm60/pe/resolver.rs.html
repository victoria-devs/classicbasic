6/// Resolves a symbol exported by a specified DLL, matching either by function name or export ordinal.
7///
8/// # Arguments
9/// * `dll_name` - Name of the DLL (e.g., `"msvbvm60.dll"`, `"kernel32.dll"`, `"user32.dll"`).
10/// * `symbol_name` - Optional symbol name string.
11/// * `ordinal` - Optional 16-bit ordinal number.
12///
13/// # Returns
14/// Virtual memory address of the resolved function.
15pub fn resolve_symbol(dll_name: &str, symbol_name: Option<&str>, ordinal: Option<u16>) -> usize {
16    let dll = dll_name.to_ascii_lowercase();
17
18    if dll.starts_with("msvbvm60") || dll.starts_with("msvbvm50") {
19        resolve_msvbvm60(symbol_name, ordinal)
20    } else if dll.starts_with("kernel32") {
21        resolve_kernel32(symbol_name, ordinal)
22    } else if dll.starts_with("user32") {
23        resolve_user32(symbol_name, ordinal)
24    } else {
25        #[cfg(feature = "std")]
26        eprintln!(
27            "[pe resolver] Warning: Unrecognized DLL '{}', returning stub",
28            dll_name
29        );
30        win32_DefaultStub as *const () as usize
31    }
32}

34/// Resolves symbols requested from `kernel32.dll`.
35fn resolve_kernel32(symbol: Option<&str>, _ordinal: Option<u16>) -> usize {
36    match symbol {
37        Some("ExitProcess") => win32_ExitProcess as *const () as usize,
38        Some("GetModuleHandleA") => win32_GetModuleHandleA as *const () as usize,
39        Some("GetModuleHandleW") => win32_GetModuleHandleW as *const () as usize,
40        Some("GetCommandLineA") => win32_GetCommandLineA as *const () as usize,
41        Some("GetCommandLineW") => win32_GetCommandLineW as *const () as usize,
42        Some("GetStartupInfoA") => win32_GetStartupInfoA as *const () as usize,
43        Some("GetVersion") => win32_GetVersion as *const () as usize,
44        Some("AllocConsole") => win32_AllocConsole as *const () as usize,
45        Some("GetStdHandle") => win32_GetStdHandle as *const () as usize,
46        Some("WriteConsoleA") => win32_WriteConsoleA as *const () as usize,
47        Some("CreateFileA") => win32_CreateFileA as *const () as usize,
48        Some("WriteFile") => win32_WriteFile as *const () as usize,
49        Some("ReadFile") => win32_ReadFile as *const () as usize,
50        Some("CloseHandle") => win32_CloseHandle as *const () as usize,
51        Some("GetFileSize") => win32_GetFileSize as *const () as usize,
52        Some("GetLastError") => win32_GetLastError as *const () as usize,
53        Some("SetLastError") => win32_SetLastError as *const () as usize,
54        Some("GetTickCount") => win32_GetTickCount as *const () as usize,
55        _ => win32_DefaultStub as *const () as usize,
56    }
57}

59/// Resolves symbols requested from `user32.dll`.
60fn resolve_user32(symbol: Option<&str>, _ordinal: Option<u16>) -> usize {
61    match symbol {
62        Some("MessageBoxA") => win32_MessageBoxA as *const () as usize,
63        _ => win32_DefaultStub as *const () as usize,
64    }
65}
