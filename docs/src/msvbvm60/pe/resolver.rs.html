6/// Resolves a symbol exported by a specified DLL, matching either by function name or export ordinal.
7///
8/// # Arguments
9/// * `dll_name` - Name of the DLL (e.g., `"msvbvm60.dll"`, `"kernel32.dll"`, `"user32.dll"`).
10/// * `symbol_name` - Optional symbol name string.
11/// * `ordinal` - Optional 16-bit ordinal number.
12///
13/// # Returns
14/// Virtual memory address of the resolved function.
15pub fn resolve_symbol(dll_name: &str, symbol_name: Option<&str>, ordinal: Option<u16>) -> usize {
16    let dll = dll_name.to_ascii_lowercase();
17
18    if dll.starts_with("msvbvm60") || dll.starts_with("msvbvm50") {
19        resolve_msvbvm60(symbol_name, ordinal)
20    } else if dll.starts_with("kernel32") {
21        resolve_kernel32(symbol_name, ordinal)
22    } else if dll.starts_with("user32") {
23        resolve_user32(symbol_name, ordinal)
24    } else {
25        #[cfg(feature = "std")]
26        eprintln!("[pe resolver] Warning: Unrecognized DLL '{}', returning stub", dll_name);
27        win32_DefaultStub as *const () as usize
28    }
29}

31/// Resolves symbols requested from `kernel32.dll`.
32fn resolve_kernel32(symbol: Option<&str>, _ordinal: Option<u16>) -> usize {
33    match symbol {
34        Some("ExitProcess") => win32_ExitProcess as *const () as usize,
35        Some("GetModuleHandleA") => win32_GetModuleHandleA as *const () as usize,
36        Some("GetModuleHandleW") => win32_GetModuleHandleW as *const () as usize,
37        Some("GetCommandLineA") => win32_GetCommandLineA as *const () as usize,
38        Some("GetCommandLineW") => win32_GetCommandLineW as *const () as usize,
39        Some("GetStartupInfoA") => win32_GetStartupInfoA as *const () as usize,
40        Some("GetVersion") => win32_GetVersion as *const () as usize,
41        Some("AllocConsole") => win32_AllocConsole as *const () as usize,
42        Some("GetStdHandle") => win32_GetStdHandle as *const () as usize,
43        Some("WriteConsoleA") => win32_WriteConsoleA as *const () as usize,
44        Some("GetLastError") => win32_GetLastError as *const () as usize,
45        Some("SetLastError") => win32_SetLastError as *const () as usize,
46        _ => win32_DefaultStub as *const () as usize,
47    }
48}

50/// Resolves symbols requested from `user32.dll`.
51fn resolve_user32(symbol: Option<&str>, _ordinal: Option<u16>) -> usize {
52    match symbol {
53        Some("MessageBoxA") => win32_MessageBoxA as *const () as usize,
54        _ => win32_DefaultStub as *const () as usize,
55    }
