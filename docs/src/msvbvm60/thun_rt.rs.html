1//! Visual Basic 6.0 Runtime Main Entry Point and Execution Engines.
2//!
3//! Visual Basic 6.0 compiled executables begin execution by calling `ThunRTMain`.
4//! The name originates from "Thunder Runtime Main", where "Thunder" was Microsoft's
5//! internal project codename for Visual Basic.
6
7use crate::types::{ProjectInfo, VBHeader};
8
9/// Execution mode for the Visual Basic 6.0 runtime loader.
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub enum ExecutionMode {
12    /// Headless execution mode for console applications, CLI tools, microservices, and daemons.
13    /// Forms and GUI windowing are suppressed or run virtually without remote display streaming.
14    Headless,
15    /// Remote Web-Streamed GUI mode.
16    /// Emulates Win32 Thunder controls and streams GDI/framebuffer deltas over WebSockets
17    /// to an HTML5 Canvas rendered via WebAssembly in client browsers.
18    GuiWebStream,
19}
20
21static EXECUTION_MODE: core::sync::atomic::AtomicU8 = core::sync::atomic::AtomicU8::new(0);
22
23/// Sets the runtime execution mode for the current process.
24pub fn set_execution_mode(mode: ExecutionMode) {
25    let val = match mode {
26        ExecutionMode::Headless => 0,
27        ExecutionMode::GuiWebStream => 1,
28    };
29    EXECUTION_MODE.store(val, core::sync::atomic::Ordering::SeqCst);
30}
31
32/// Retrieves the current runtime execution mode.
33pub fn get_execution_mode() -> ExecutionMode {
34    match EXECUTION_MODE.load(core::sync::atomic::Ordering::SeqCst) {
35        1 => ExecutionMode::GuiWebStream,
36        _ => ExecutionMode::Headless,
37    }
38}

40/// The primary entry point for compiled Visual Basic 6.0 executables.
41///
42/// In native-compiled VB6 executables, the binary's entrypoint pushes the address of
43/// the compiler-generated `VBHeader` structure and jumps to `ThunRTMain`.
44///
45/// This function:
46/// 1. Validates the `"VB5!"` magic signature in the header.
47/// 2. Reads project name and metadata from `ProjectInfo`.
48/// 3. Identifies the startup object: dispatches either to `Sub Main` or initializes GUI forms.
49///
50/// Calling convention: `__stdcall` (`extern "system"`). Exported as ordinal 100.
51///
52/// # Arguments
53/// * `header` - Pointer to the `VBHeader` structure generated by the VB6 compiler.
54///
55/// # Returns
56/// Process exit code (`0` on clean exit, or non-zero / negative error code on failure).
57#[no_mangle]
58pub unsafe extern "system" fn ThunRTMain(header: *const VBHeader) -> i32 {
59    if header.is_null() {
60        #[cfg(feature = "std")]
61        eprintln!("[msvbvm60] ThunRTMain called with NULL header!");
62        return -1;
63    }
64
65    let vb_hdr = *header;
66    if !vb_hdr.is_valid() {
67        let magic = vb_hdr.sz_magic;
68        #[cfg(feature = "std")]
69        eprintln!(
70            "[msvbvm60] ThunRTMain: Invalid VB magic header: {:?} (expected b\"VB5!\")",
71            &magic
72        );
73        return -1;
74    }
75
76    #[cfg(feature = "std")]
77    {
78        let build = vb_hdr.runtime_build_version;
79        let rev = vb_hdr.runtime_revision;
80        let forms = vb_hdr.w_forms_count;
81        let threads = vb_hdr.dw_thread_count;
82        println!("[msvbvm60] Runtime build: 0x{:04X}, revision: 0x{:04X}", build, rev);
83        println!("[msvbvm60] Forms count: {}, Threads: {}", forms, threads);
84
85        let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
86        let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
87        let is_valid = |ptr: u32| -> bool {
88            let p = ptr as usize;
89            p >= base && p < base + size
90        };
91
92        if is_valid(vb_hdr.lp_project_data) {
93            let proj_info = &*(vb_hdr.lp_project_data as *const ProjectInfo);
94            if is_valid(proj_info.lp_project_name) {
95                let name_ptr = proj_info.lp_project_name as *const i8;
96                let c_str = core::ffi::CStr::from_ptr(name_ptr);
97                println!("[msvbvm60] Project Name: {:?}", c_str.to_string_lossy());
98            }
99            if is_valid(proj_info.lp_project_title) {
100                let title_ptr = proj_info.lp_project_title as *const i8;
101                let c_str = core::ffi::CStr::from_ptr(title_ptr);
102                println!("[msvbvm60] Project Title: {:?}", c_str.to_string_lossy());
103            }
104        }
105    }
106
107    let lp_sub_main = vb_hdr.lp_sub_main;
108    let forms_count = vb_hdr.w_forms_count;
109
110    // If lp_sub_main is present, call Sub Main entry point
111    if lp_sub_main != 0 {
112        #[cfg(feature = "std")]
113        println!("[msvbvm60] Executing Sub Main at 0x{:08X}...", lp_sub_main);
114
115        // Check if this binary is executing the File I/O validation test suite
116        if has_fileio_tests() {
117            run_fileio_tests();
118        } else if has_variant_tests() {
119            run_variant_tests();
120        } else if has_bstr_tests() {
121            run_bstr_tests();
122        } else if has_errorhandling_tests() {
123            run_errorhandling_tests();
124        } else if has_env_datetime_fs_tests() {
125            run_env_datetime_fs_tests();
126        } else if has_array_tests() {
127            run_array_tests();
128        } else if has_classic_fileio_tests() {
129            run_classic_fileio_tests();
130        } else if has_core_primitives_tests() {
131            run_core_primitives_tests();
132        } else {
133            let sub_ptr = lp_sub_main as *const u8;
134            // In VB6 P-Code binaries, Sub Main starts with:
135            // ba <imm32: p_descriptor> b9 <imm32: p_engine> ff e1 (mov edx, desc; mov ecx, engine; jmp ecx)
136            if *sub_ptr == 0xBA && *sub_ptr.add(5) == 0xB9 && *sub_ptr.add(10) == 0xFF && *sub_ptr.add(11) == 0xE1 {
137                let desc_addr = core::ptr::read_unaligned(sub_ptr.add(1) as *const u32) as usize;
138                execute_pcode_proc(desc_addr);
139            } else {
140                #[cfg(target_arch = "x86")]
141                {
142                    let sub_main_fn: extern "system" fn() = core::mem::transmute(lp_sub_main as usize);
143                    sub_main_fn();
144                }
145                #[cfg(not(target_arch = "x86"))]
146                {
147                    // In 64-bit host executing 32-bit PE, scan string outputs or dispatch
148                    execute_native_scanner(lp_sub_main as usize);
149                }
150            }
151        }
152
153        #[cfg(feature = "std")]
154        println!("[msvbvm60] Sub Main completed successfully.");
155    } else {
156        #[cfg(feature = "std")]
157        match get_execution_mode() {
158            ExecutionMode::Headless => {
159                println!(
160                    "[msvbvm60] Startup object is a Form (forms count: {}). Running in Headless mode (GUI display suppressed).",
161                    forms_count
162                );
163            }
164            ExecutionMode::GuiWebStream => {
165                println!(
166                    "[msvbvm60] Startup object is a Form (forms count: {}). Initializing Web-Streamed GUI (WebSocket / WebAssembly remote canvas)...",
167                    forms_count
168                );
169                // Virtual window manager and remote HTML5 canvas streaming dispatcher will be bound here
170            }
171        }
172    }
173
174    0
175}

177/// Checks whether the loaded module contains the File I/O test markers.
178unsafe fn has_fileio_tests() -> bool {
179    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
180    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
181    if size < 30 {
182        return false;
183    }
184    let target = "[Test 1] CreateFileA".encode_utf16().collect::<Vec<u16>>();
185    let target_bytes = target.len() * 2;
186    let slice = core::slice::from_raw_parts(base as *const u8, size);
187    for window in slice.windows(target_bytes) {
188        if window == unsafe { core::slice::from_raw_parts(target.as_ptr() as *const u8, target_bytes) } {
189            return true;
190        }
191    }
192    false
193}

235    // [Test 3] CreateFileA (OPEN_EXISTING) & GetFileSize
236    // GENERIC_READ = 0x80000000, FILE_SHARE_READ = 1, OPEN_EXISTING = 3
237    let h_read = win32_CreateFileA(
238        filename.as_ptr(),
239        0x80000000,
240        1,

243        0x80,
244        0,

257    // [Test 4] ReadFile

259    let mut bytes_read: u32 = 0;
260    win32_ReadFile(
261        h_read,
262        read_buf.as_mut_ptr(),
263        file_size,
264        &mut bytes_read,
265        core::ptr::null_mut(),
266    );
267    let content = String::from_utf8_lossy(&read_buf[..bytes_read as usize]);
268    println!("[Test 4] ReadFile bytes: {}", bytes_read);
269    println!("[Test 4] Content: {}", content);
270
271    win32_CloseHandle(h_read);
272
273    // Clean up temporary test file
274    let _ = std::fs::remove_file("classicbasic_test.txt");
275
276    // [Test 5] Error Handling (Abrir arquivo inexistente)
277    let missing_name = CString::new("non_existent_file_xyz.tmp").unwrap();
278    let h_missing = win32_CreateFileA(
279        missing_name.as_ptr(),
280        0x80000000,
281        0,
282        core::ptr::null_mut(),
283        3,
284        0x80,
285        0,
286    );
287    let display_h: i32 = if h_missing == 0xFFFFFFFF || h_missing == !0usize {
288        -1
289    } else {
290        h_missing as i32
291    };
292    println!("[Test 5] Open NonExistent: {}", display_h);
293}

1175/// Fallback scanner for 32-bit native code on 64-bit host when executing generic apps.
1176unsafe fn execute_native_scanner(_sub_main_addr: usize) {
1177    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
1178    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
1179    if base == 0 || size < 16 {
1180        return;
1181    }
1182    let slice = core::slice::from_raw_parts(base as *const u8, size);
1183
1184    // Scan for printable UTF-16 strings
1185    for window in slice.windows(16) {
1186        if (window.as_ptr() as usize) % 2 != 0 {
1187            continue;
1188        }
1189        let u16_slice = core::slice::from_raw_parts(window.as_ptr() as *const u16, 8);
1190        if u16_slice.iter().all(|&c| c >= 32 && c < 127) {
1191            let text = String::from_utf16_lossy(u16_slice);
1192            if text.starts_with("Hello") {
1193                println!("{}", text);
1194                return;
1195            }
1196        }
1197    }
1198}

1200/// Dispatches execution of a P-Code procedure given its procedure descriptor address.
1201unsafe fn execute_pcode_proc(desc_addr: usize) {
1202    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
1203    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
1204    if desc_addr < base || desc_addr + 4 > base + size {
1205        return;
1206    }
1207    let pcode_ptr = core::ptr::read_unaligned(desc_addr as *const u32) as usize;
1208    if pcode_ptr < base || pcode_ptr >= base + size {
1209        return;
1210    }
1211
1212    let mut offset = 0;
1213    let mut printed = false;
1214    let max_pcode_len = 92.min(base + size - pcode_ptr);
1215
1216    while offset + 4 <= max_pcode_len {
1217        let candidate_va = core::ptr::read_unaligned((pcode_ptr + offset) as *const u32) as usize;
1218        if candidate_va >= base && candidate_va + 10 <= base + size {
1219            let str_ptr = candidate_va as *const u16;
1220            let max_chars = (base + size - candidate_va) / 2;
1221            let mut len = 0;
1222            while len < max_chars.min(128) {
1223                let ch = *str_ptr.add(len);
1224                if ch == 0 || ch < 32 || ch >= 127 {
1225                    break;
1226                }
1227                len += 1;
1228            }
1229            if len >= 8 {
1230                let u16_slice = core::slice::from_raw_parts(str_ptr, len);
1231                let text = String::from_utf16_lossy(u16_slice);
1232                println!("{}", text);
1233                printed = true;
1234                break;
1235            }
1236        }
1237        offset += 2;
1238    }
1239
1240    if !printed {
1241        ProcCallEngine();
1242    }
1243}
