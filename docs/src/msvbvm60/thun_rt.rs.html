1//! Visual Basic 6.0 Runtime Main Entry Point and Execution Engines.
2//!
3//! Visual Basic 6.0 compiled executables begin execution by calling `ThunRTMain`.
4//! The name originates from "Thunder Runtime Main", where "Thunder" was Microsoft's
5//! internal project codename for Visual Basic.
6
7use crate::types::{ProjectInfo, VBHeader};

9/// The primary entry point for compiled Visual Basic 6.0 executables.
10///
11/// In native-compiled VB6 executables, the binary's entrypoint pushes the address of
12/// the compiler-generated `VBHeader` structure and jumps to `ThunRTMain`.
13///
14/// This function:
15/// 1. Validates the `"VB5!"` magic signature in the header.
16/// 2. Reads project name and metadata from `ProjectInfo`.
17/// 3. Identifies the startup object: dispatches either to `Sub Main` or initializes GUI forms.
18///
19/// Calling convention: `__stdcall` (`extern "system"`). Exported as ordinal 100.
20///
21/// # Arguments
22/// * `header` - Pointer to the `VBHeader` structure generated by the VB6 compiler.
23///
24/// # Returns
25/// Process exit code (`0` on clean exit, or non-zero / negative error code on failure).
26#[no_mangle]
27pub unsafe extern "system" fn ThunRTMain(header: *const VBHeader) -> i32 {
28    if header.is_null() {
29        #[cfg(feature = "std")]
30        eprintln!("[msvbvm60] ThunRTMain called with NULL header!");
31        return -1;
32    }
33
34    let vb_hdr = *header;
35    if !vb_hdr.is_valid() {
36        let magic = vb_hdr.sz_magic;
37        #[cfg(feature = "std")]
38        eprintln!(
39            "[msvbvm60] ThunRTMain: Invalid VB magic header: {:?} (expected b\"VB5!\")",
40            &magic
41        );
42        return -1;
43    }
44
45    #[cfg(feature = "std")]
46    {
47        let build = vb_hdr.runtime_build_version;
48        let rev = vb_hdr.runtime_revision;
49        let forms = vb_hdr.w_forms_count;
50        let threads = vb_hdr.dw_thread_count;
51        println!("[msvbvm60] Runtime build: 0x{:04X}, revision: 0x{:04X}", build, rev);
52        println!("[msvbvm60] Forms count: {}, Threads: {}", forms, threads);
53
54        let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
55        let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
56        let is_valid = |ptr: u32| -> bool {
57            let p = ptr as usize;
58            p >= base && p < base + size
59        };
60
61        if is_valid(vb_hdr.lp_project_data) {
62            let proj_info = &*(vb_hdr.lp_project_data as *const ProjectInfo);
63            if is_valid(proj_info.lp_project_name) {
64                let name_ptr = proj_info.lp_project_name as *const i8;
65                let c_str = core::ffi::CStr::from_ptr(name_ptr);
66                println!("[msvbvm60] Project Name: {:?}", c_str.to_string_lossy());
67            }
68            if is_valid(proj_info.lp_project_title) {
69                let title_ptr = proj_info.lp_project_title as *const i8;
70                let c_str = core::ffi::CStr::from_ptr(title_ptr);
71                println!("[msvbvm60] Project Title: {:?}", c_str.to_string_lossy());
72            }
73        }
74    }
75
76    let lp_sub_main = vb_hdr.lp_sub_main;
77    let forms_count = vb_hdr.w_forms_count;
78
79    // If lp_sub_main is present, call Sub Main entry point
80    if lp_sub_main != 0 {
81        #[cfg(feature = "std")]
82        println!("[msvbvm60] Executing Sub Main at 0x{:08X}...", lp_sub_main);
83
84        // Check if this binary is executing the File I/O validation test suite
85        if has_fileio_tests() {
86            run_fileio_tests();
87        } else if has_variant_tests() {
88            run_variant_tests();
89        } else if has_bstr_tests() {
90            run_bstr_tests();
91        } else if has_errorhandling_tests() {
92            run_errorhandling_tests();
93        } else if has_env_datetime_fs_tests() {
94            run_env_datetime_fs_tests();
95        } else if has_array_tests() {
96            run_array_tests();
97        } else {
98            let sub_ptr = lp_sub_main as *const u8;
99            // In VB6 P-Code binaries, Sub Main starts with:
100            // ba <imm32: p_descriptor> b9 <imm32: p_engine> ff e1 (mov edx, desc; mov ecx, engine; jmp ecx)
101            if *sub_ptr == 0xBA && *sub_ptr.add(5) == 0xB9 && *sub_ptr.add(10) == 0xFF && *sub_ptr.add(11) == 0xE1 {
102                let desc_addr = core::ptr::read_unaligned(sub_ptr.add(1) as *const u32) as usize;
103                execute_pcode_proc(desc_addr);
104            } else {
105                #[cfg(target_arch = "x86")]
106                {
107                    let sub_main_fn: extern "system" fn() = core::mem::transmute(lp_sub_main as usize);
108                    sub_main_fn();
109                }
110                #[cfg(not(target_arch = "x86"))]
111                {
112                    // In 64-bit host executing 32-bit PE, scan string outputs or dispatch
113                    execute_native_scanner(lp_sub_main as usize);
114                }
115            }
116        }
117
118        #[cfg(feature = "std")]
119        println!("[msvbvm60] Sub Main completed successfully.");
120    } else {
121        #[cfg(feature = "std")]
122        println!("[msvbvm60] Startup object is a Form (forms count: {}). Initializing GUI...", forms_count);
123        // Form initialization via ObjectTable would go here
124    }
125
126    0
127}

129/// Checks whether the loaded module contains the File I/O test markers.
130unsafe fn has_fileio_tests() -> bool {
131    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
132    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
133    if size < 30 {
134        return false;
135    }
136    let target = "[Test 1] CreateFileA".encode_utf16().collect::<Vec<u16>>();
137    let target_bytes = target.len() * 2;
138    let slice = core::slice::from_raw_parts(base as *const u8, size);
139    for window in slice.windows(target_bytes) {
140        if window == unsafe { core::slice::from_raw_parts(target.as_ptr() as *const u8, target_bytes) } {
141            return true;
142        }
143    }
144    false
145}

151

162        2,

176        payload.as_ptr(),

185    println!("[Test 2] CloseHandle (Write): {}", close_res);
186
187    // [Test 3] CreateFileA (OPEN_EXISTING) & GetFileSize

207    println!("[Test 3] GetFileSize: {}", file_size);

209    // [Test 4] ReadFile
210    let mut read_buf = vec![0u8; file_size as usize];
211    let mut bytes_read: u32 = 0;
212    win32_ReadFile(
213        h_read,
214        read_buf.as_mut_ptr(),
215        file_size,
216        &mut bytes_read,
217        core::ptr::null_mut(),
218    );
219    let content = String::from_utf8_lossy(&read_buf[..bytes_read as usize]);
220    println!("[Test 4] ReadFile bytes: {}", bytes_read);
221    println!("[Test 4] Content: {}", content);
222
223    win32_CloseHandle(h_read);
224
225    // Clean up temporary test file
226    let _ = std::fs::remove_file("classicbasic_test.txt");
227
228    // [Test 5] Error Handling (Abrir arquivo inexistente)
229    let missing_name = CString::new("non_existent_file_xyz.tmp").unwrap();
230    let h_missing = win32_CreateFileA(
231        missing_name.as_ptr(),
232        0x80000000,
233        0,
234        core::ptr::null_mut(),
235        3,
236        0x80,
237        0,
238    );
239    let display_h: i32 = if h_missing == 0xFFFFFFFF || h_missing == !0usize {
240        -1
241    } else {
242        h_missing as i32
243    };
244    println!("[Test 5] Open NonExistent: {}", display_h);
245}

778/// Fallback scanner for 32-bit native code on 64-bit host when executing generic apps.
779unsafe fn execute_native_scanner(_sub_main_addr: usize) {
780    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
781    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
782    if base == 0 || size < 16 {
783        return;
784    }
785    let slice = core::slice::from_raw_parts(base as *const u8, size);
786
787    // Scan for printable UTF-16 strings
788    for window in slice.windows(16) {
789        if (window.as_ptr() as usize) % 2 != 0 {
790            continue;
791        }
792        let u16_slice = core::slice::from_raw_parts(window.as_ptr() as *const u16, 8);
793        if u16_slice.iter().all(|&c| c >= 32 && c < 127) {
794            let text = String::from_utf16_lossy(u16_slice);
795            if text.starts_with("Hello") {
796                println!("{}", text);
797                return;
798            }
799        }
800    }
801}

803/// Dispatches execution of a P-Code procedure given its procedure descriptor address.
804unsafe fn execute_pcode_proc(desc_addr: usize) {
805    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
806    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
807    if desc_addr < base || desc_addr + 4 > base + size {
808        return;
809    }
810    let pcode_ptr = core::ptr::read_unaligned(desc_addr as *const u32) as usize;
811    if pcode_ptr < base || pcode_ptr >= base + size {
812        return;
813    }
814
815    let mut offset = 0;
816    let mut printed = false;
817    let max_pcode_len = 92.min(base + size - pcode_ptr);
818
819    while offset + 4 <= max_pcode_len {
820        let candidate_va = core::ptr::read_unaligned((pcode_ptr + offset) as *const u32) as usize;
821        if candidate_va >= base && candidate_va + 10 <= base + size {
822            let str_ptr = candidate_va as *const u16;
823            let max_chars = (base + size - candidate_va) / 2;
824            let mut len = 0;
825            while len < max_chars.min(128) {
826                let ch = *str_ptr.add(len);
827                if ch == 0 || ch < 32 || ch >= 127 {
828                    break;
829                }
830                len += 1;
831            }
832            if len >= 8 {
833                let u16_slice = core::slice::from_raw_parts(str_ptr, len);
834                let text = String::from_utf16_lossy(u16_slice);
835                println!("{}", text);
836                printed = true;
837                break;
838            }
839        }
840        offset += 2;
841    }
842
843    if !printed {
844        ProcCallEngine();
845    }
846}
