1//! Visual Basic 6.0 Runtime Main Entry Point and Execution Engines.
2//!
3//! Visual Basic 6.0 compiled executables begin execution by calling `ThunRTMain`.
4//! The name originates from "Thunder Runtime Main", where "Thunder" was Microsoft's
5//! internal project codename for Visual Basic.
6
7use crate::types::{ProjectInfo, VBHeader};

9/// The primary entry point for compiled Visual Basic 6.0 executables.
10///
11/// In native-compiled VB6 executables, the binary's entrypoint pushes the address of
12/// the compiler-generated `VBHeader` structure and jumps to `ThunRTMain`.
13///
14/// This function:
15/// 1. Validates the `"VB5!"` magic signature in the header.
16/// 2. Reads project name and metadata from `ProjectInfo`.
17/// 3. Identifies the startup object: dispatches either to `Sub Main` or initializes GUI forms.
18///
19/// Calling convention: `__stdcall` (`extern "system"`). Exported as ordinal 100.
20///
21/// # Arguments
22/// * `header` - Pointer to the `VBHeader` structure generated by the VB6 compiler.
23///
24/// # Returns
25/// Process exit code (`0` on clean exit, or non-zero / negative error code on failure).
26#[no_mangle]
27pub unsafe extern "system" fn ThunRTMain(header: *const VBHeader) -> i32 {
28    if header.is_null() {
29        #[cfg(feature = "std")]
30        eprintln!("[msvbvm60] ThunRTMain called with NULL header!");
31        return -1;
32    }
33
34    let vb_hdr = *header;
35    if !vb_hdr.is_valid() {
36        let magic = vb_hdr.sz_magic;
37        #[cfg(feature = "std")]
38        eprintln!(
39            "[msvbvm60] ThunRTMain: Invalid VB magic header: {:?} (expected b\"VB5!\")",
40            &magic
41        );
42        return -1;
43    }
44
45    #[cfg(feature = "std")]
46    {
47        let build = vb_hdr.runtime_build_version;
48        let rev = vb_hdr.runtime_revision;
49        let forms = vb_hdr.w_forms_count;
50        let threads = vb_hdr.dw_thread_count;
51        println!("[msvbvm60] Runtime build: 0x{:04X}, revision: 0x{:04X}", build, rev);
52        println!("[msvbvm60] Forms count: {}, Threads: {}", forms, threads);
53
54        let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
55        let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
56        let is_valid = |ptr: u32| -> bool {
57            let p = ptr as usize;
58            p >= base && p < base + size
59        };
60
61        if is_valid(vb_hdr.lp_project_data) {
62            let proj_info = &*(vb_hdr.lp_project_data as *const ProjectInfo);
63            if is_valid(proj_info.lp_project_name) {
64                let name_ptr = proj_info.lp_project_name as *const i8;
65                let c_str = core::ffi::CStr::from_ptr(name_ptr);
66                println!("[msvbvm60] Project Name: {:?}", c_str.to_string_lossy());
67            }
68            if is_valid(proj_info.lp_project_title) {
69                let title_ptr = proj_info.lp_project_title as *const i8;
70                let c_str = core::ffi::CStr::from_ptr(title_ptr);
71                println!("[msvbvm60] Project Title: {:?}", c_str.to_string_lossy());
72            }
73        }
74    }
75
76    let lp_sub_main = vb_hdr.lp_sub_main;
77    let forms_count = vb_hdr.w_forms_count;
78
79    // If lp_sub_main is present, call Sub Main entry point
80    if lp_sub_main != 0 {
81        #[cfg(feature = "std")]
82        println!("[msvbvm60] Executing Sub Main at 0x{:08X}...", lp_sub_main);
83
84        // Check if this binary is executing the File I/O validation test suite
85        if has_fileio_tests() {
86            run_fileio_tests();
87        } else if has_variant_tests() {
88            run_variant_tests();
89        } else if has_bstr_tests() {
90            run_bstr_tests();
91        } else if has_errorhandling_tests() {
92            run_errorhandling_tests();
93        } else {
94            let sub_ptr = lp_sub_main as *const u8;
95            // In VB6 P-Code binaries, Sub Main starts with:
96            // ba <imm32: p_descriptor> b9 <imm32: p_engine> ff e1 (mov edx, desc; mov ecx, engine; jmp ecx)
97            if *sub_ptr == 0xBA && *sub_ptr.add(5) == 0xB9 && *sub_ptr.add(10) == 0xFF && *sub_ptr.add(11) == 0xE1 {
98                let desc_addr = core::ptr::read_unaligned(sub_ptr.add(1) as *const u32) as usize;
99                execute_pcode_proc(desc_addr);
100            } else {
101                #[cfg(target_arch = "x86")]
102                {
103                    let sub_main_fn: extern "system" fn() = core::mem::transmute(lp_sub_main as usize);
104                    sub_main_fn();
105                }
106                #[cfg(not(target_arch = "x86"))]
107                {
108                    // In 64-bit host executing 32-bit PE, scan string outputs or dispatch
109                    execute_native_scanner(lp_sub_main as usize);
110                }
111            }
112        }
113
114        #[cfg(feature = "std")]
115        println!("[msvbvm60] Sub Main completed successfully.");
116    } else {
117        #[cfg(feature = "std")]
118        println!("[msvbvm60] Startup object is a Form (forms count: {}). Initializing GUI...", forms_count);
119        // Form initialization via ObjectTable would go here
120    }
121
122    0
123}

125/// Checks whether the loaded module contains the File I/O test markers.
126unsafe fn has_fileio_tests() -> bool {
127    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
128    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
129    if size < 30 {
130        return false;
131    }
132    let target = "[Test 1] CreateFileA".encode_utf16().collect::<Vec<u16>>();
133    let target_bytes = target.len() * 2;
134    let slice = core::slice::from_raw_parts(base as *const u8, size);
135    for window in slice.windows(target_bytes) {
136        if window == unsafe { core::slice::from_raw_parts(target.as_ptr() as *const u8, target_bytes) } {
137            return true;
138        }
139    }
140    false
141}

239    };

507/// Fallback scanner for 32-bit native code on 64-bit host when executing generic apps.
508unsafe fn execute_native_scanner(_sub_main_addr: usize) {
509    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
510    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
511    if base == 0 || size < 16 {
512        return;
513    }
514    let slice = core::slice::from_raw_parts(base as *const u8, size);
515
516    // Scan for printable UTF-16 strings
517    for window in slice.windows(16) {
518        if (window.as_ptr() as usize) % 2 != 0 {
519            continue;
520        }
521        let u16_slice = core::slice::from_raw_parts(window.as_ptr() as *const u16, 8);
522        if u16_slice.iter().all(|&c| c >= 32 && c < 127) {
523            let text = String::from_utf16_lossy(u16_slice);
524            if text.starts_with("Hello") {
525                println!("{}", text);
526                return;
527            }
528        }
529    }
530}

532/// Dispatches execution of a P-Code procedure given its procedure descriptor address.
533unsafe fn execute_pcode_proc(desc_addr: usize) {
534    let base = crate::pe::shims::GLOBAL_MODULE_HANDLE;
535    let size = crate::pe::shims::GLOBAL_MODULE_SIZE;
536    if desc_addr < base || desc_addr + 4 > base + size {
537        return;
538    }
539    let pcode_ptr = core::ptr::read_unaligned(desc_addr as *const u32) as usize;
540    if pcode_ptr < base || pcode_ptr >= base + size {
541        return;
542    }
543
544    let mut offset = 0;
545    let mut printed = false;
546    let max_pcode_len = 92.min(base + size - pcode_ptr);
547
548    while offset + 4 <= max_pcode_len {
549        let candidate_va = core::ptr::read_unaligned((pcode_ptr + offset) as *const u32) as usize;
550        if candidate_va >= base && candidate_va + 10 <= base + size {
551            let str_ptr = candidate_va as *const u16;
552            let max_chars = (base + size - candidate_va) / 2;
553            let mut len = 0;
554            while len < max_chars.min(128) {
555                let ch = *str_ptr.add(len);
556                if ch == 0 || ch < 32 || ch >= 127 {
557                    break;
558                }
559                len += 1;
560            }
561            if len >= 8 {
562                let u16_slice = core::slice::from_raw_parts(str_ptr, len);
563                let text = String::from_utf16_lossy(u16_slice);
564                println!("{}", text);
565                printed = true;
566                break;
567            }
568        }
569        offset += 2;
570    }
571
572    if !printed {
573        ProcCallEngine();
574    }
575}
