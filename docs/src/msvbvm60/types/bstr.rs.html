1//! COM Automation BSTR representation.
2//!
3//! A BSTR is a pointer to the first UTF-16 code unit of a string.
4//! The 4 bytes directly preceding the pointer contain the byte length
5//! of the string (excluding the null terminator). The string is also
6//! terminated by a null UTF-16 character (0x0000).
7
8pub type Bstr = *mut u16;
9
10/// Returns the byte length of a BSTR by inspecting the 4 bytes preceding the pointer.
11pub unsafe fn bstr_byte_len(s: Bstr) -> u32 {
12    if s.is_null() {
13        return 0;
14    }
15    let len_ptr = (s as *const u8).sub(4) as *const u32;
16    *len_ptr
17}
18
19/// Returns the number of UTF-16 code units in the BSTR.
20pub unsafe fn bstr_len(s: Bstr) -> usize {
21    (bstr_byte_len(s) / 2) as usize
22}
23
24/// Converts a BSTR to a Rust slice of u16 code units.
25pub unsafe fn bstr_as_slice<'a>(s: Bstr) -> &'a [u16] {
26    if s.is_null() {
27        return &[];
28    }
29    let len = bstr_len(s);
30    core::slice::from_raw_parts(s, len)
31}
