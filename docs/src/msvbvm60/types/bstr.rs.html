1//! COM Automation BSTR representation and heap allocation engine.
2//!
3//! A BSTR ("Basic String") is a pointer to the first UTF-16 code unit of a string.
4//! The 4 bytes directly preceding the pointer contain the byte length
5//! of the string (excluding the null terminator). The string is also
6//! terminated by a null UTF-16 character (0x0000).
7
8use std::alloc::{alloc, dealloc, Layout};
9
10/// OLE Automation BSTR pointer type representing a pointer to a null-terminated UTF-16 buffer.
11pub type Bstr = *mut u16;
12
13/// Returns the byte length of a BSTR by inspecting the 4 bytes preceding the pointer.
14///
15/// # Arguments
16/// * `s` - BSTR pointer.
17///
18/// # Safety
19/// If non-null, `s` must be a valid BSTR allocated with 4 leading length bytes.
20pub unsafe fn bstr_byte_len(s: Bstr) -> u32 {
21    if s.is_null() {
22        return 0;
23    }
24    let len_ptr = (s as *const u8).sub(4) as *const u32;
25    *len_ptr
26}
27
28/// Returns the number of UTF-16 code units in the BSTR.
29///
30/// # Arguments
31/// * `s` - BSTR pointer.
32///
33/// # Safety
34/// If non-null, `s` must be a valid BSTR.
35pub unsafe fn bstr_len(s: Bstr) -> usize {
36    (bstr_byte_len(s) / 2) as usize
37}

39/// Converts a BSTR to a Rust slice of u16 code units.
40///
41/// # Arguments
42/// * `s` - BSTR pointer.
43///
44/// # Safety
45/// If non-null, `s` must point to valid memory for the duration of the slice lifetime `'a`.
46pub unsafe fn bstr_as_slice<'a>(s: Bstr) -> &'a [u16] {
47    if s.is_null() {
48        return &[];
49    }
50    let len = bstr_len(s);
51    core::slice::from_raw_parts(s, len)
52}

54/// Allocates a new BSTR with `len` UTF-16 code units, initialized from `str_in` if non-null.
55///
56/// # Arguments
57/// * `str_in` - Optional pointer to source UTF-16 code units.
58/// * `len` - Number of UTF-16 characters to allocate.
59///
60/// # Returns
61/// Pointer to the newly allocated BSTR, or null if allocation fails.
62///
63/// # Safety
64/// `str_in` must be valid for reading `len` code units if non-null.
65#[no_mangle]
66pub unsafe extern "system" fn SysAllocStringLen(str_in: *const u16, len: u32) -> Bstr {
67    let byte_len = len * 2;
68    // Layout: 4-byte prefix + byte_len bytes + 2-byte null terminator
69    let total_bytes = 4 + (byte_len as usize) + 2;
70    let layout = match Layout::from_size_align(total_bytes, 4) {
71        Ok(l) => l,
72        Err(_) => return core::ptr::null_mut(),
73    };
74
75    let mem = alloc(layout);
76    if mem.is_null() {
77        return core::ptr::null_mut();
78    }
79
80    *(mem as *mut u32) = byte_len;
81    let bstr_ptr = mem.add(4) as *mut u16;
82
83    if !str_in.is_null() {
84        core::ptr::copy_nonoverlapping(str_in, bstr_ptr, len as usize);
85    } else {
86        core::ptr::write_bytes(bstr_ptr, 0, (len as usize) * 2);
87    }
88    *bstr_ptr.add(len as usize) = 0; // UTF-16 null terminator
89
90    bstr_ptr
91}

93/// Allocates a new BSTR from a null-terminated UTF-16 string.
94///
95/// # Arguments
96/// * `str_in` - Pointer to null-terminated UTF-16 string.
97///
98/// # Returns
99/// Pointer to the newly allocated BSTR, or null if `str_in` is null.
100///
101/// # Safety
102/// `str_in` must point to a valid null-terminated sequence of `u16` code units.
103#[no_mangle]
104pub unsafe extern "system" fn SysAllocString(str_in: *const u16) -> Bstr {
105    if str_in.is_null() {
106        return core::ptr::null_mut();
107    }
108    let mut len = 0;
109    while *str_in.add(len) != 0 {
110        len += 1;
111    }
112    SysAllocStringLen(str_in, len as u32)
113}

115/// Deallocates a BSTR string previously allocated via SysAllocString* or runtime helpers.
116///
117/// # Arguments
118/// * `bstr` - Pointer to the BSTR string to release.
119///
120/// # Safety
121/// `bstr` must have been allocated via `SysAllocStringLen` / `SysAllocString`.
122#[no_mangle]
123pub unsafe extern "system" fn SysFreeString(bstr: Bstr) {
124    if bstr.is_null() {
125        return;
126    }
127    let mem = (bstr as *mut u8).sub(4);
128    let byte_len = *(mem as *const u32);
129    let total_bytes = 4 + (byte_len as usize) + 2;
130    if let Ok(layout) = Layout::from_size_align(total_bytes, 4) {
131        dealloc(mem, layout);
132    }
133}

159/// Helper function to allocate a BSTR from a Rust `&str`.
160///
161/// # Arguments
162/// * `s` - Rust string slice.
163///
164/// # Returns
165/// Pointer to the newly allocated BSTR.
166pub fn sys_alloc_string(s: &str) -> Bstr {
167    let utf16: Vec<u16> = s.encode_utf16().collect();
168    unsafe { SysAllocStringLen(utf16.as_ptr(), utf16.len() as u32) }
169}

171/// Helper function to convert a BSTR into an owned Rust `String`.
172///
173/// # Arguments
174/// * `s` - BSTR pointer.
175///
176/// # Safety
177/// `s` must be valid or null.
178pub unsafe fn bstr_to_string(s: Bstr) -> String {
179    if s.is_null() {
180        return String::new();
181    }
182    let slice = bstr_as_slice(s);
183    String::from_utf16_lossy(slice)
184}

186/// Converts a null-terminated UTF-16 pointer into an owned Rust `String`.
187///
188/// # Safety
189/// If non-null, `p` must point to a null-terminated UTF-16 array.
190pub unsafe fn utf16_ptr_to_string(p: *const u16) -> String {
191    if p.is_null() {
192        return String::new();
193    }
194    let mut len = 0;
195    while *p.add(len) != 0 {
196        len += 1;
197    }
198    String::from_utf16_lossy(core::slice::from_raw_parts(p, len))
199}
