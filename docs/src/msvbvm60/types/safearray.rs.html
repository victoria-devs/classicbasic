1//! COM and Visual Basic 6.0 SAFEARRAY structures.
2//!
3//! Visual Basic 6 arrays are represented in memory as OLE Automation `SAFEARRAY` structures,
4//! storing dimension counts, feature flags, lock counts, and boundary descriptors (`SAFEARRAYBOUND`).
5
6/// Array allocated on the stack.
7pub const FADF_AUTO: u16 = 0x0001;
8/// Array statically allocated.
9pub const FADF_STATIC: u16 = 0x0002;
10/// Array embedded in a structure or record.
11pub const FADF_EMBEDDED: u16 = 0x0004;
12/// Array cannot be resized or reallocated.
13pub const FADF_FIXEDSIZE: u16 = 0x0010;
14/// Array contains User Defined Type (UDT) records.
15pub const FADF_RECORD: u16 = 0x0020;
16/// Array contains interface pointers and has an associated IID.
17pub const FADF_HAVEIID: u16 = 0x0040;
18/// Array has an associated `VarType` identifying element type.
19pub const FADF_HAVEVARTYPE: u16 = 0x0080;
20/// Array contains BSTR elements.
21pub const FADF_BSTR: u16 = 0x0100;
22/// Array contains `IUnknown` interface pointers.
23pub const FADF_UNKNOWN: u16 = 0x0200;
24/// Array contains `IDispatch` interface pointers.
25pub const FADF_DISPATCH: u16 = 0x0400;
26/// Array contains `Variant` elements.
27pub const FADF_VARIANT: u16 = 0x0800;
28
29/// Boundary descriptor for a single dimension of a `SafeArray`.
30#[repr(C)]
31#[derive(Copy, Clone, Debug)]
32pub struct SafeArrayBound {
33    /// Number of elements in this dimension.
34    pub c_elements: u32,
35    /// Lower bound index for this dimension (e.g. `0` or `1`).
36    pub l_lbound: i32,
37}
38
39/// OLE Automation `SAFEARRAY` descriptor structure.
40#[repr(C)]
41#[derive(Copy, Clone, Debug)]
42pub struct SafeArray {
43    /// Count of dimensions.
44    pub c_dims: u16,
45    /// Flags describing array memory layout and element features (`FADF_*`).
46    pub f_features: u16,
47    /// Size in bytes of an individual array element.
48    pub cb_elements: u32,
49    /// Lock counter tracking how many active references are locking the array.
50    pub c_locks: u32,
51    /// Pointer to the raw array element data.
52    pub pv_data: *mut core::ffi::c_void,
53    /// Flexible array member containing dimension boundary descriptors.
54    pub rgsabound: [SafeArrayBound; 1],
55}
56
57impl SafeArray {
58    /// Creates and allocates a 1-dimensional `SafeArray` vector.
59    pub unsafe fn create_vector(
60        elem_size: u32,
61        lbound: i32,
62        c_elements: u32,
63        features: u16,
64    ) -> *mut SafeArray {
65        let total_header_size = core::mem::size_of::<SafeArray>();
66        let data_bytes = (c_elements as usize) * (elem_size as usize);
67
68        let header_layout = std::alloc::Layout::from_size_align(total_header_size, 8).unwrap();
69        let p_sa = std::alloc::alloc_zeroed(header_layout) as *mut SafeArray;
70        if p_sa.is_null() {
71            return core::ptr::null_mut();
72        }
73
74        let data_layout = std::alloc::Layout::from_size_align(data_bytes.max(1), 8).unwrap();
75        let p_data = std::alloc::alloc_zeroed(data_layout) as *mut core::ffi::c_void;
76
77        (*p_sa).c_dims = 1;
78        (*p_sa).f_features = features;
79        (*p_sa).cb_elements = elem_size;
80        (*p_sa).c_locks = 0;
81        (*p_sa).pv_data = p_data;
82        (*p_sa).rgsabound[0] = SafeArrayBound {
83            c_elements,
84            l_lbound: lbound,
85        };
86
87        p_sa
88    }
89
90    /// Redimensions an existing `SafeArray`, preserving existing elements.
91    pub unsafe fn redim_preserve(p_sa: *mut SafeArray, new_ubound: i32, new_lbound: i32) {
92        if p_sa.is_null() {
93            return;
94        }
95        let old_elements = (*p_sa).rgsabound[0].c_elements as usize;
96        let elem_size = (*p_sa).cb_elements as usize;
97        let new_elements = if new_ubound >= new_lbound {
98            (new_ubound - new_lbound + 1) as usize
99        } else {
100            0
101        };
102
103        let new_data_bytes = new_elements * elem_size;
104        let new_layout = std::alloc::Layout::from_size_align(new_data_bytes.max(1), 8).unwrap();
105        let new_data = std::alloc::alloc_zeroed(new_layout) as *mut core::ffi::c_void;
106
107        if !(*p_sa).pv_data.is_null() {
108            let copy_count = old_elements.min(new_elements);
109            if copy_count > 0 {
110                core::ptr::copy_nonoverlapping((*p_sa).pv_data, new_data, copy_count * elem_size);
111            }
112            let old_data_bytes = old_elements * elem_size;
113            let old_layout = std::alloc::Layout::from_size_align(old_data_bytes.max(1), 8).unwrap();
114            std::alloc::dealloc((*p_sa).pv_data as *mut u8, old_layout);
115        }
116
117        (*p_sa).pv_data = new_data;
118        (*p_sa).rgsabound[0].c_elements = new_elements as u32;
119        (*p_sa).rgsabound[0].l_lbound = new_lbound;
120    }
121
122    /// Deallocates the element buffer and the `SafeArray` descriptor.
123    pub unsafe fn destroy(p_sa: *mut SafeArray) {
124        if p_sa.is_null() {
125            return;
126        }
127        if !(*p_sa).pv_data.is_null() {
128            let data_bytes = ((*p_sa).rgsabound[0].c_elements as usize) * ((*p_sa).cb_elements as usize);
129            let data_layout = std::alloc::Layout::from_size_align(data_bytes.max(1), 8).unwrap();
130            std::alloc::dealloc((*p_sa).pv_data as *mut u8, data_layout);
131            (*p_sa).pv_data = core::ptr::null_mut();
132        }
133        let header_layout = std::alloc::Layout::from_size_align(core::mem::size_of::<SafeArray>(), 8).unwrap();
134        std::alloc::dealloc(p_sa as *mut u8, header_layout);
135    }
