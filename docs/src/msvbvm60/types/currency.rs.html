1//! Visual Basic 6.0 Currency (CY) data type.
2//!
3//! Visual Basic's `Currency` is an 8-byte (64-bit) two's complement integer,
4//! scaled by 10,000 to give a fixed-point number with 15 digits to the left
5//! of the decimal point and 4 digits to the right.
6
7/// Visual Basic 6.0 64-bit fixed-point Currency (`CY`) structure.
8#[repr(C)]
9#[derive(Copy, Clone, Default, PartialEq, Eq, PartialOrd, Ord, Debug)]
10pub struct Currency(pub i64);
11
12impl Currency {
13    /// Fixed scale factor (10,000) representing four decimal digits.
14    pub const SCALE: i64 = 10_000;
15
16    /// Constructs a `Currency` from a pre-scaled 64-bit raw integer value.
17    ///
18    /// # Arguments
19    /// * `val` - 64-bit integer already multiplied by 10,000.
20    ///
21    /// # Returns
22    /// Initialized `Currency` instance.
23    pub fn from_scaled(val: i64) -> Self {
24        Self(val)
25    }
26
27    /// Converts a floating-point number into a fixed-point `Currency` by multiplying by 10,000.
28    ///
29    /// # Arguments
30    /// * `val` - 64-bit float value.
31    ///
32    /// # Returns
33    /// Initialized `Currency` instance.
34    pub fn from_f64(val: f64) -> Self {
35        Self((val * Self::SCALE as f64) as i64)
36    }
37
38    /// Converts the fixed-point `Currency` value into a 64-bit floating-point number.
39    ///
40    /// # Returns
41    /// 64-bit float representation.
42    pub fn to_f64(self) -> f64 {
43        self.0 as f64 / Self::SCALE as f64
44    }
45}
