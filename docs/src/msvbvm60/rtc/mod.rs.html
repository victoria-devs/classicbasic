10/// Implementation of VB6 `RGB(Red, Green, Blue)`.
11///
12/// Combines red, green, and blue color values into a 32-bit Win32 `COLORREF` integer.
13///
14/// # Arguments
15/// * `red` - Red color component (0 to 255).
16/// * `green` - Green color component (0 to 255).
17/// * `blue` - Blue color component (0 to 255).
18///
19/// # Returns
20/// 32-bit integer formatted as `(Red) | (Green << 8) | (Blue << 16)`.
21#[no_mangle]
22pub unsafe extern "system" fn rtcRgb(red: i32, green: i32, blue: i32) -> i32 {
23    let r = (red.clamp(0, 255) as u32) & 0xFF;
24    let g = (green.clamp(0, 255) as u32) & 0xFF;
25    let b = (blue.clamp(0, 255) as u32) & 0xFF;
26    (r | (g << 8) | (b << 16)) as i32
27}

171/// Optimized character-based implementation of VB6 `Left$(String, Length)`.
172///
173/// # Arguments
174/// * `bstr` - Source BSTR string.
175/// * `len` - Number of characters to extract.
176///
177/// # Returns
178/// Extracted BSTR substring.
179#[no_mangle]
180pub unsafe extern "system" fn rtcLeftCharBstr(bstr: Bstr, len: i32) -> Bstr {
181    if bstr.is_null() || len <= 0 {
182        return sys_alloc_string("");
183    }
184    let s = bstr_as_slice(bstr);
185    let take_count = (len as usize).min(s.len());
186    SysAllocStringLen(s.as_ptr(), take_count as u32)
187}

223/// Implementation of VB6 `Mid(String, Start, [Length])`.
224///
225/// Returns a substring from a string starting at character position `Start`.
226///
227/// # Arguments
228/// * `bstr` - Source BSTR string.
229/// * `start` - 1-based start position.
230/// * `len_var` - Number of characters to extract (passed as Variant pointer or null).
231///
232/// # Returns
233/// Extracted BSTR substring.
234#[no_mangle]
235pub unsafe extern "system" fn rtcMidCharBstr(
236    bstr: Bstr,
237    start: i32,
238    len_var: *const Variant,
239) -> Bstr {
240    if bstr.is_null() || start <= 0 {
241        return sys_alloc_string("");
242    }
243    let s = bstr_as_slice(bstr);
244    let start_idx = (start as usize) - 1;
245    if start_idx >= s.len() {
246        return sys_alloc_string("");
247    }
248    let remaining = s.len() - start_idx;
249    let take_count = if !len_var.is_null() {
250        if let Some(l) = (*len_var).as_i32() {
251            if l < 0 {
252                remaining
253            } else {
254                (l as usize).min(remaining)
255            }
256        } else {
257            remaining
258        }
259    } else {
260        remaining
261    };
262    SysAllocStringLen(s[start_idx..].as_ptr(), take_count as u32)
263}

276    if bstr.is_null() {
