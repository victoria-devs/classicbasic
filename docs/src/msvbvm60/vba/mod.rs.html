1//! Visual Basic Architecture (`__vba*`) compiler runtime helpers.
2//!
3//! When the Visual Basic 6 native-code compiler (C2.EXE) compiles functions,
4//! it generates calls to `__vba*` helpers for string management, variant arithmetic,
5//! array lifecycle, and SEH (Structured Exception Handling).

17/// Copies a block of memory from source to destination.
18///
19/// Emitted by the compiler for fixed-size buffer copying, UDT (User Defined Type)
20/// assignments, and memory transfers.
21///
22/// # Arguments
23/// * `count` - Number of bytes to copy.
24/// * `dst` - Pointer to destination buffer.
25/// * `src` - Pointer to source buffer.
26#[no_mangle]
27pub unsafe extern "system" fn __vbaCopyBytes(count: usize, dst: *mut u8, src: *const u8) {
28    if !dst.is_null() && !src.is_null() && count > 0 {
29        core::ptr::copy(src, dst, count);
30    }
31}
32
33/// Copies memory from source to destination, zero-filling the destination if the source is null.
34///
35/// # Arguments
36/// * `count` - Number of bytes to copy or zero.
37/// * `dst` - Pointer to destination buffer.
38/// * `src` - Pointer to source buffer, or null to zero-fill `dst`.
39#[no_mangle]
40pub unsafe extern "system" fn __vbaCopyBytesZero(count: usize, dst: *mut u8, src: *const u8) {
41    if !dst.is_null() && count > 0 {
42        if !src.is_null() {
43            core::ptr::copy(src, dst, count);
44        } else {
45            core::ptr::write_bytes(dst, 0, count);
46        }
47    }
48}

54/// Assigns a BSTR to a destination pointer variable by copying the string reference.
55///
56/// # Arguments
57/// * `dst` - Pointer to destination BSTR variable.
58/// * `src` - Source BSTR pointer.
59///
60/// # Returns
61/// The resulting BSTR pointer.
62#[no_mangle]
63pub unsafe extern "system" fn __vbaStrCopy(dst: *mut Bstr, src: Bstr) -> Bstr {
64    if !dst.is_null() {
65        if !(*dst).is_null() {
66            SysFreeString(*dst);
67        }
68        if !src.is_null() {
69            let slice = bstr_as_slice(src);
70            *dst = SysAllocStringLen(slice.as_ptr(), slice.len() as u32);
71        } else {
72            *dst = core::ptr::null_mut();
73        }
74        return *dst;
75    }
76    src
77}

79/// Moves a BSTR from source to destination, transferring ownership and clearing source.
80///
81/// # Arguments
82/// * `dst` - Pointer to destination BSTR variable.
83/// * `src` - Pointer to source BSTR variable (set to null on move).
84///
85/// # Returns
86/// The moved BSTR pointer.
87#[no_mangle]
88pub unsafe extern "system" fn __vbaStrMove(dst: *mut Bstr, src: *mut Bstr) -> Bstr {
89    if !dst.is_null() {
90        if !(*dst).is_null() {
91            SysFreeString(*dst);
92        }
93        if !src.is_null() {
94            *dst = *src;
95            *src = core::ptr::null_mut();
96        } else {
97            *dst = core::ptr::null_mut();
98        }
99        return *dst;
100    }
101    core::ptr::null_mut()
102}

104/// Concatenates two BSTR strings and returns a newly allocated BSTR pointer.
105///
106/// # Arguments
107/// * `s1` - First BSTR string.
108/// * `s2` - Second BSTR string to concatenate.
109///
110/// # Returns
111/// The concatenated BSTR pointer.
112#[no_mangle]
113pub unsafe extern "system" fn __vbaStrCat(s1: Bstr, s2: Bstr) -> Bstr {
114    let sl1 = if !s1.is_null() { bstr_as_slice(s1) } else { &[] };
115    let sl2 = if !s2.is_null() { bstr_as_slice(s2) } else { &[] };
116    let total_len = sl1.len() + sl2.len();
117    let mut buf = Vec::with_capacity(total_len);
118    buf.extend_from_slice(sl1);
119    buf.extend_from_slice(sl2);
120    SysAllocStringLen(buf.as_ptr(), total_len as u32)
121}

177/// Searches for a substring within a string (equivalent to VB6 `InStr`).
178///
179/// # Arguments
180/// * `arg1` - Start index or comparison method flag.
181/// * `arg2` - Pointer to haystack or needle BSTR.
182/// * `arg3` - Pointer to needle or haystack BSTR.
183/// * `arg4` - Start index or comparison method flag.
184///
185/// # Returns
186/// 1-based character index of the match, or 0 if not found.
187#[no_mangle]
188pub unsafe extern "system" fn __vbaInStr(
189    arg1: i32,
190    arg2: usize,
191    arg3: usize,
192    arg4: i32,
193) -> i32 {
194    let b1 = arg2 as Bstr;
195    let b2 = arg3 as Bstr;
196    let (str_haystack, str_needle) = if bstr_len(b1) >= bstr_len(b2) {
197        (b1, b2)
198    } else {
199        (b2, b1)
200    };
201    if str_haystack.is_null() || str_needle.is_null() {
202        return 0;
203    }
204    let hay = bstr_as_slice(str_haystack);
205    let needle = bstr_as_slice(str_needle);
206    if needle.is_empty() {
207        return 1;
208    }
209    let start_pos = if arg1 > 0 { arg1 } else if arg4 > 0 { arg4 } else { 1 };
210    let start_idx = (start_pos.max(1) - 1) as usize;
211    if start_idx >= hay.len() {
212        return 0;
213    }
214    for i in start_idx..=hay.len().saturating_sub(needle.len()) {
215        if &hay[i..i + needle.len()] == needle {
216            return (i + 1) as i32;
217        }
218    }
219    0
220}

222/// Converts an ANSI string buffer to a Unicode BSTR.
223///
224/// # Arguments
225/// * `dst` - Pointer to destination BSTR variable.
226/// * `ansi_src` - Pointer to null-terminated ANSI string buffer.
227///
228/// # Returns
229/// The newly allocated Unicode BSTR.
230#[no_mangle]
231pub unsafe extern "system" fn __vbaStrToUnicode(dst: *mut Bstr, ansi_src: *const u8) -> Bstr {
232    if ansi_src.is_null() {
233        if !dst.is_null() {
234            *dst = core::ptr::null_mut();
235        }
236        return core::ptr::null_mut();
237    }
238    let mut len = 0;
239    while *ansi_src.add(len) != 0 {
240        len += 1;
241    }
242    let slice = core::slice::from_raw_parts(ansi_src, len);
243    let s = String::from_utf8_lossy(slice);
244    let b = sys_alloc_string(&s);
245    if !dst.is_null() {
246        if !(*dst).is_null() {
247            SysFreeString(*dst);
248        }
249        *dst = b;
250    }
251    b
252}

254/// Converts a Unicode BSTR to an ANSI string buffer.
255///
256/// # Arguments
257/// * `dst` - Pointer to destination ANSI buffer pointer.
258/// * `unicode_src` - Source Unicode BSTR string.
259///
260/// # Returns
261/// Pointer to the newly allocated ANSI character buffer.
262#[no_mangle]
263pub unsafe extern "system" fn __vbaStrToAnsi(dst: *mut *mut u8, unicode_src: Bstr) -> *mut u8 {
264    if unicode_src.is_null() {
265        if !dst.is_null() {
266            *dst = core::ptr::null_mut();
267        }
268        return core::ptr::null_mut();
269    }
270    let s = bstr_to_string(unicode_src);
271    let bytes = s.into_bytes();
272    let total_len = bytes.len() + 1;
273    let layout = std::alloc::Layout::from_size_align(total_len, 1).unwrap();
274    let mem = std::alloc::alloc(layout);
275    core::ptr::copy_nonoverlapping(bytes.as_ptr(), mem, bytes.len());
276    *mem.add(bytes.len()) = 0; // null terminator
277    if !dst.is_null() {
278        *dst = mem;
279    }
280    mem
281}

283/// Compares two BSTR strings lexicographically.
284///
285/// # Arguments
286/// * `s1` - First BSTR string.
287/// * `s2` - Second BSTR string.
288///
289/// # Returns
290/// * `< 0` if `s1` is less than `s2`.
291/// * `0` if `s1` equals `s2`.
292/// * `> 0` if `s1` is greater than `s2`.
293#[no_mangle]
294pub unsafe extern "system" fn __vbaStrCmp(s1: Bstr, s2: Bstr) -> i32 {
295    if s1 == s2 {
296        return 0;
297    }
298    if s1.is_null() {
299        return -1;
300    }
301    if s2.is_null() {
302        return 1;
303    }
304    let sl1 = bstr_as_slice(s1);
305    let sl2 = bstr_as_slice(s2);
306    for (c1, c2) in sl1.iter().zip(sl2.iter()) {
307        if c1 != c2 {
308            return (*c1 as i32) - (*c2 as i32);
309        }
310    }
311    (sl1.len() as i32) - (sl2.len() as i32)
312}

408/// Dynamically adds two `Variant` values, performing numeric addition or string concatenation.
409///
410/// # Arguments
411/// * `dst` - Pointer to destination `Variant`.
412/// * `right` - Right operand `Variant`.
413/// * `left` - Left operand `Variant`.
414///
415/// # Returns
416/// Pointer to destination `Variant`.
417#[no_mangle]
418pub unsafe extern "system" fn __vbaVarAdd(
419    dst: *mut Variant,
420    right: *const Variant,
421    left: *const Variant,
422) -> *mut Variant {
423    if dst.is_null() || left.is_null() || right.is_null() {
424        return dst;
425    }
426    if (*left).vt != VT_BSTR && (*right).vt != VT_BSTR {
427        if let (Some(l), Some(r)) = ((*left).as_i32(), (*right).as_i32()) {
428            *dst = Variant::from_i32(l.wrapping_add(r));
429            return dst;
430        }
431    }
432    if let (Some(fl), Some(fr)) = ((*left).as_f64(), (*right).as_f64()) {
433        let sum = fl + fr;
434        if sum.fract() == 0.0 && sum.abs() <= i32::MAX as f64 && sum.abs() >= i32::MIN as f64 {
435            *dst = Variant::from_i32(sum as i32);
436        } else {
437            *dst = Variant::from_f64(sum);
438        }
439    } else {
440        let sl = (*left).to_string_representation();
441        let sr = (*right).to_string_representation();
442        let combined = format!("{}{}", sl, sr);
443        *dst = Variant::from_bstr(sys_alloc_string(&combined));
444    }
445    dst
446}

662/// Dynamically performs logical or bitwise NOT on a `Variant` value (`Not src`).
663///
664/// # Arguments
665/// * `dst` - Pointer to destination `Variant`.
666/// * `src` - Source `Variant` operand.
667///
668/// # Returns
669/// Pointer to destination `Variant`.
670#[no_mangle]
671pub unsafe extern "system" fn __vbaVarNot(
672    dst: *mut Variant,
673    src: *const Variant,
674) -> *mut Variant {
675    if dst.is_null() || src.is_null() {
676        return dst;
677    }
678    if (*src).vt == VT_BOOL {
679        let b = (*src).as_bool().unwrap_or(false);

685    }
